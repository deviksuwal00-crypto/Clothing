    <script>
        let gearList = [];
        
        // Hide/Show Cart
        function toggleCart() { 
            document.getElementById('cart').classList.toggle('translate-x-full'); 
        }

        // Add to List
        function addToList(name, price) {
            gearList.push({ name, price });
            updateUI();
            
            // Show a quick alert so they know it worked
            alert(name + " added to your list!");
        }

        // Update Cart Interface
        function updateUI() {
            const listItems = document.getElementById('cart-items');
            const totalDisplay = document.getElementById('total-price');
            const count = document.getElementById('count');
            const shipping = parseInt(document.getElementById('shipping').value);

            if(gearList.length === 0) {
                listItems.innerHTML = '<p class="text-white/30 italic">No gear selected yet.</p>';
            } else {
                listItems.innerHTML = gearList.map((item, index) => `
                    <div class="flex justify-between items-center bg-[#111] p-4 rounded-lg">
                        <span>${item.name}</span>
                        <div class="flex items-center gap-4">
                            <span class="text-green-500 font-bold">Rs. ${item.price}</span>
                            <button onclick="removeItem(${index})" class="text-red-500 text-xs font-bold">X</button>
                        </div>
                    </div>
                `).join('');
            }
            
            const subtotal = gearList.reduce((a, b) => a + b.price, 0);
            totalDisplay.innerText = `NPR ${subtotal + shipping}`;
            count.innerText = gearList.length;
        }

        // Remove item from list
        function removeItem(index) {
            gearList.splice(index, 1);
            updateUI();
        }

        function updateTotal() { updateUI(); }

        // WhatsApp Checkout
        function checkout() {
            if(gearList.length === 0) return alert("Your list is empty!");
            
            let message = `*Hello Hunting Cloths!* I want to order:%0A%0A`;
            gearList.forEach(item => message += `- ${item.name} (Rs. ${item.price})%0A`);
            
            const shippingType = document.getElementById('shipping').options[document.getElementById('shipping').selectedIndex].text;
            message += `%0A*Shipping:* ${shippingType}`;
            message += `%0A*Total:* ${document.getElementById('total-price').innerText}`;
            
            // Put your actual phone number here
            window.location.href = `https://wa.me/977XXXXXXXXXX?text=${message}`;
        }
